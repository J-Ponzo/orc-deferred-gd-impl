#version 450

layout(location = 0) out vec4 frag_color;

layout(set = 0, binding = 0) uniform Globals {
    vec2 screen_size;
    vec2 _pad; // 16 bytes align
} globals;

layout(set = 0, binding = 1) uniform sampler2D shaded_tex;

#include "res://orc-renderers-impl/orc-deferred-gd-impl/shaders/shader_utils.glslinc.txt"

void main() {
	vec2 uv = (gl_FragCoord.xy) / globals.screen_size;
	vec3 shaded_color = texture(shaded_tex, uv).rgb;
    frag_color = vec4(linear_to_srgb(shaded_color), 1.0);
}

#version 450

layout(location = 0) in vec3 v_world_position;  // TODO : Less computations if we do this in view space

layout(set = 1, binding = 0) uniform LightParams { 
	vec3 light_position;                        // TODO : useless if we go view space
    float light_range;
};

void main() {
    float distance_to_light = length(v_world_position - light_position);
    gl_FragDepth = distance_to_light / light_range;
}
